## Quick Python / Pygame Loader for All 10 Characters

Since every `*_4dir_sheet.png` is standardized to `1000x674` (`250x674` per frame), one helper function loads your entire 10-character cast:

```python
import pygame
from pathlib import Path

FRAME_WIDTH = 250
FRAME_HEIGHT = 674
DIRECTIONS = ["bottom_right", "bottom_left", "top_right", "top_left"]

CHARACTER_SHEETS = {
    "businessman": "steampunk_businessman_4dir_sheet.png",
    "student_daughter": "steampunk_student_daughter_4dir_sheet.png",
    "student_son": "steampunk_student_son_4dir_sheet.png",
    "church_sister": "steampunk_church_sister_4dir_sheet.png",
    "chief_stoker": "steampunk_chief_stoker_4dir_sheet.png",
    "inspector": "steampunk_inspector_4dir_sheet.png",
    "constable": "steampunk_constable_4dir_sheet.png",
    "apothecary": "steampunk_apothecary_4dir_sheet.png",
    "professor": "steampunk_professor_4dir_sheet.png",
    "student_friend": "steampunk_student_friend_4dir_sheet.png",
}

def load_isometric_character(sheet_path: str | Path, target_height: int = 96) -> dict[str, pygame.Surface]:
    sheet = pygame.image.load(str(sheet_path)).convert_alpha()
    scale_factor = target_height / FRAME_HEIGHT
    target_width = int(FRAME_WIDTH * scale_factor)

    sprites = {}
    for i, direction in enumerate(DIRECTIONS):
        rect = pygame.Rect(i * FRAME_WIDTH, 0, FRAME_WIDTH, FRAME_HEIGHT)
        frame = sheet.subsurface(rect).copy()
        sprites[direction] = pygame.transform.scale(frame, (target_width, target_height))
    return sprites

def load_all_characters(assets_dir: str | Path, target_height: int = 96) -> dict[str, dict[str, pygame.Surface]]:
    base = Path(assets_dir)
    return {
        name: load_isometric_character(base / filename, target_height)
        for name, filename in CHARACTER_SHEETS.items()
    }
```
